POST /v1/shifts

  -H "Authorization: Bearer sk_live_..." \
  -H "Content-Type: application/json" \
  -d '{
    "worker_id": "EMP-123",
    "location_id": "LOC-456",
    "scheduled_start": "2025-01-14T07:00:00Z",

POST /v1/timesheets
Submit timesheet data from your time tracking system
curl -X POST https://api.hypertrack.com/v1/timesheets \
  -H "Authorization: Bearer sk_live_..." \

GET /v1/reconciled-timesheets
Retrieve all reconciled timesheets with confidence scores and payouts
curl https://api.hypertrack.com/v1/reconciled-timesheets \
  -H "Authorization: Bearer sk_live_..." \
  -G \
  -d "date=2025-01-14" \
  -d "status=pending_approval"
GET /v1/recommended-payouts
Get payout recommendations based on reconciled data
curl https://api.hypertrack.com/v1/recommended-payouts \
  -H "Authorization: Bearer sk_live_..." \
  -G \
  -d "worker_id=EMP-123" \
  -d "date_from=2025-01-07" \
  -d "date_to=2025-01-14"
